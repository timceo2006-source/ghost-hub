-- 2. ตัวแปรควบคุม
local autoFarmActive = false
local loopTask = nil
local heartbeatConnection = nil
local TweenService = game:GetService("TweenService")
local isTweening = false 

-- [ฟังก์ชันหาโฟลเดอร์ตามเลเวลและเกาะ]
local function GetTargetFolder(level)
    local zones = workspace:FindFirstChild("NPC Zones")
    if not zones then return nil end
    local islandName = (level >= 40) and "Clown Island" or "Starter Island"
    return zones:FindFirstChild(islandName) and zones[islandName]:FindFirstChild("NPCS")
end

local function runFarmLogic()
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local Player = Players.LocalPlayer
    local lockedTarget = nil

    -- [ตั้งค่าเควสและจุดบินรอ]
    -- ใส่พิกัด Vector3.new(x, y, z) ของจุดที่จะให้ไปบินรอที่เกาะนั้นๆ
    local function getQuestInfo()
        local level = Player.PlayerData.Experience.Level.Value 
        if level >= 60 then return "Level 60", "Clown Boss", "2", Vector3.new(0, 100, 0) -- แก้เลขตรงนี้
        elseif level >= 40 then return "Level 40", "Pirate Clown", "2", Vector3.new(0, 100, 0) -- แก้เลขตรงนี้
        elseif level >= 25 then return "Level 25", "Bandit Boss", "1", Vector3.new(0, 100, 0)
        else return "Level 1", "Thief", "1", Vector3.new(0, 100, 0) end
    end

    local function acceptQuest(questName, giverIndex)
        local NpcWorkspace = workspace:FindFirstChild("Npc_Workspace")
        local QuestGiver = NpcWorkspace and NpcWorkspace:FindFirstChild("QuestGivers") and NpcWorkspace.QuestGivers:FindFirstChild(giverIndex)
        local questFunc = Player:FindFirstChild("PlayerGui") and Player.PlayerGui:FindFirstChild("QuestGui") and Player.PlayerGui.QuestGui:FindFirstChild("QuestFunction")
        if questFunc and QuestGiver then pcall(function() questFunc:InvokeServer(QuestGiver, questName) end) end
    end

    -- ลูปสมอง
    loopTask = task.spawn(function()
        while autoFarmActive do
            local Character = Player.Character
            if Character and Character:FindFirstChild("Humanoid") and Character.Humanoid.Health > 0 then
                local currentLevel = Player.PlayerData.Experience.Level.Value
                local TargetFolder = GetTargetFolder(currentLevel)

                if TargetFolder then
                    local qName, _, gIndex = getQuestInfo()
                    if Player.Quest.GiverName.Value == "" then acceptQuest(qName, gIndex) end

                    if not (lockedTarget and lockedTarget:FindFirstChild("Humanoid") and lockedTarget.Humanoid.Health > 0) then
                        lockedTarget = nil
                        local _, tName = getQuestInfo() -- ดึงชื่อมอน
                        for _, obj in pairs(TargetFolder:GetChildren()) do
                            local isTarget = string.match(obj.Name, "^" .. tName)
                            local isBoss = string.find(string.lower(obj.Name), "boss") ~= nil
                            
                            if isTarget and obj:FindFirstChild("Humanoid") and obj.Humanoid.Health > 0 then
                                -- Logic การเลือกมอน
                                if currentLevel >= 60 then
                                    if isBoss then lockedTarget = obj; break end
                                elseif currentLevel >= 40 then
                                    if not isBoss then lockedTarget = obj; break end
                                else
                                    if (currentLevel < 25 and not isBoss) or (currentLevel >= 25 and isBoss) then
                                        lockedTarget = obj; break
                                    end
                                end
                            end
                        end
                    end
                end
            end
            task.wait(0.5)
        end
    end)

    -- ลูปเคลื่อนที่ (บิน/ลอยตัว)
    heartbeatConnection = RunService.Heartbeat:Connect(function()
        if not autoFarmActive then return end
        local Character = Player.Character
        local HRP = Character and Character:FindFirstChild("HumanoidRootPart")
        local _, _, _, hoverPos = getQuestInfo() -- ดึงจุดที่จะไปบินรอ

        if HRP then
            Character.Humanoid:ChangeState(Enum.HumanoidStateType.Physics) -- บังคับลอยตัว
            
            if lockedTarget and lockedTarget:FindFirstChild("HumanoidRootPart") then
                -- มีมอน: พุ่งไปหา
                local targetPos = lockedTarget.HumanoidRootPart.Position + Vector3.new(0, 5, 0)
                if (HRP.Position - targetPos).Magnitude > 15 then
                    if not isTweening then
                        isTweening = true
                        TweenService:Create(HRP, TweenInfo.new(0.5, Enum.EasingStyle.Linear), {CFrame = CFrame.new(targetPos)}):Play()
                        task.wait(0.6); isTweening = false
                    end
                else
                    HRP.CFrame = CFrame.lookAt(targetPos, lockedTarget.HumanoidRootPart.Position)
                    local tool = Character:FindFirstChild("Combat") or Player.Backpack:FindFirstChild("Combat")
                    if tool then 
                        if not Character:FindFirstChild("Combat") then Character.Humanoid:EquipTool(tool) end
                        tool:Activate() 
                    end
                end
            else
                -- ไม่มีมอน: บินไปรอที่จุด Hover
                if (HRP.Position - hoverPos).Magnitude > 5 then
                    if not isTweening then
                        isTweening = true
                        TweenService:Create(HRP, TweenInfo.new(1.5, Enum.EasingStyle.Linear), {CFrame = CFrame.new(hoverPos)}):Play()
                        task.wait(1.6); isTweening = false
                    end
                end
            end
        end
    end)
end

-- 3. Toggle
Tab:Toggle({
    Title = "AutoFarm",
    Desc = "AutoFarmLevel",
    Value = false,
    Callback = function(state)
        autoFarmActive = state
        if state then runFarmLogic()
        else
            if loopTask then task.cancel(loopTask) end
            if heartbeatConnection then heartbeatConnection:Disconnect() end
            local char = game:GetService("Players").LocalPlayer.Character
            if char and char:FindFirstChild("Humanoid") then char.Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp) end
        end
    end
})
