local Window = WindUI:CreateWindow({
    Title = "Ghost Hub",
    Icon = "door-open",
    Author = "TIM",
    Folder = "MyGhostHub",
    Size = UDim2.fromOffset(580, 460),
    Theme = "Dark",
})

local autoFarmActive = false
local loopTask = nil
local heartbeatConnection = nil
local TweenService = game:GetService("TweenService") -- เพิ่ม TweenService

-- [ส่วนเพิ่ม] ฟังก์ชันหาโฟลเดอร์มอนสเตอร์ตามเลเวล
local function GetTargetFolder(level)
    local zones = workspace:FindFirstChild("NPC Zones")
    if not zones then return nil end
    
    local islandName = "Starter Island" -- Default
    if level >= 40 then 
        islandName = "Desert Island"    -- เปลี่ยนชื่อให้ตรงกับ Folder ในเกม
    elseif level >= 25 then
        islandName = "Starter Island"   -- สมมติว่าเป็นชื่อเดิม
    end
    
    return zones:FindFirstChild(islandName) and zones[islandName]:FindFirstChild("NPCS")
end

local function runFarmLogic()
    local Players = game:GetService("Players")
    local RunService = game:GetService("RunService")
    local Player = Players.LocalPlayer
    local lockedTarget = nil

    -- (ฟังก์ชันเดิมคงไว้)
    local function getQuestInfo()
        local level = Player.PlayerData.Experience.Level.Value 
        if level >= 40 then return "Level 40", "Desert Bandit" -- ตัวอย่างชื่อมอน LV 40
        elseif level >= 25 then return "Level 25", "Bandit Boss"
        elseif level >= 10 then return "Level 10", "Bandit"
        else return "Level 1", "Thief" end
    end

    local function acceptQuest(questName)
        local NpcWorkspace = workspace:FindFirstChild("Npc_Workspace")
        local QuestGiver = NpcWorkspace and NpcWorkspace:FindFirstChild("QuestGivers") and NpcWorkspace.QuestGivers:FindFirstChild("1")
        local questFunc = Player:FindFirstChild("PlayerGui") and Player.PlayerGui:FindFirstChild("QuestGui") and Player.PlayerGui.QuestGui:FindFirstChild("QuestFunction")
        if questFunc and QuestGiver then pcall(function() questFunc:InvokeServer(QuestGiver, questName) end) end
    end

    loopTask = task.spawn(function()
        while autoFarmActive do
            local Character = Player.Character
            if Character and Character:FindFirstChild("Humanoid") and Character.Humanoid.Health > 0 then
                local currentLevel = Player.PlayerData.Experience.Level.Value
                local TargetFolder = GetTargetFolder(currentLevel) -- ใช้ฟังก์ชันใหม่

                if TargetFolder then
                    local _, targetName = getQuestInfo()
                    if Player.Quest.GiverName.Value == "" then acceptQuest(select(1, getQuestInfo())) end

                    if not (lockedTarget and lockedTarget:FindFirstChild("Humanoid") and lockedTarget.Humanoid.Health > 0) then
                        lockedTarget = nil
                        for _, obj in pairs(TargetFolder:GetChildren()) do
                            -- logic การหาเป้าหมายเดิม
                            local isTarget = string.match(obj.Name, "^" .. targetName)
                            local isBoss = string.find(string.lower(obj.Name), "boss") ~= nil
                            if isTarget and obj:FindFirstChild("Humanoid") and obj.Humanoid.Health > 0 then
                                if (currentLevel < 25 and not isBoss) or (currentLevel >= 25 and isBoss) then
                                    lockedTarget = obj; break
                                end
                            end
                        end
                    end
                end
            end
            task.wait(0.5)
        end
    end)

    heartbeatConnection = RunService.Heartbeat:Connect(function()
        if not autoFarmActive then return end
        local Character = Player.Character
        local HRP = Character and Character:FindFirstChild("HumanoidRootPart")
        local tool = Character and (Character:FindFirstChild("Combat") or Player.Backpack:FindFirstChild("Combat"))

        if lockedTarget and lockedTarget:FindFirstChild("HumanoidRootPart") and HRP then
            local targetPos = lockedTarget.HumanoidRootPart.Position + Vector3.new(0, 7, 0)
            local distance = (HRP.Position - targetPos).Magnitude

            -- [ส่วนเพิ่ม] เช็คระยะห่าง ถ้าไกลให้ Tween ถ้าใกล้ให้ Lock
            if distance > 20 then
                -- Tween ไปหา (ความเร็วปรับตามระยะทาง)
                local tweenInfo = TweenInfo.new(distance / 200, Enum.EasingStyle.Linear)
                TweenService:Create(HRP, tweenInfo, {CFrame = CFrame.new(targetPos)}):Play()
            else
                -- Lock ตำแหน่ง (เหมือนเดิม)
                Character.Humanoid:ChangeState(Enum.HumanoidStateType.Physics)
                HRP.CFrame = CFrame.lookAt(targetPos, lockedTarget.HumanoidRootPart.Position)
                HRP.AssemblyLinearVelocity = Vector3.zero
                
                if tool then
                    if not Character:FindFirstChild("Combat") then Character.Humanoid:EquipTool(tool) end
                    tool:Activate()
                end
            end
        elseif Character and Character:FindFirstChild("Humanoid") then
            Character.Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
    end)
end


-- Callback
MainTab:Toggle({
    Title = "AutoFarm",
    Desc = "AutoFarmLevel",
    Value = false,
    Callback = function(state)
        autoFarmActive = state
        if state then
            runFarmLogic()
        else
            if loopTask then task.cancel(loopTask) end
            if heartbeatConnection then heartbeatConnection:Disconnect() end
            local char = game:GetService("Players").LocalPlayer.Character
            if char and char:FindFirstChild("Humanoid") then char.Humanoid:ChangeState(Enum.HumanoidStateType.GettingUp) end
        end
    end
})
